const decode = (s: string) =>
  s.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/g, "$1")
    .replace(/&#(\d+);/g, (_, n) => String.fromCharCode(+n))
    .replace(/&#x([0-9a-f]+);/gi, (_, n) => String.fromCharCode(parseInt(n, 16)))
    .replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&quot;/g, '"').replace(/&apos;/g, "'").replace(/&nbsp;/g, " ")
    .replace(/&amp;/g, "&");
const text = (h: string) => decode(h.replace(/<[^>]*>/g, " ")).replace(/\s+/g, " ").trim();

async function fromMyfxbook(): Promise<Ev[]> {
  const r = await fetch("https://www.myfxbook.com/rss/forex-economic-calendar-events", { headers: { ...BROWSER, Referer: "https://www.myfxbook.com/forex-economic-calendar" } });
  if (!r.ok) throw new Error("myfxbook " + r.status);
  const raw = await r.text();
  const out: Ev[] = [];
  for (const it of raw.split(/<item>/i).slice(1)) {
    const g = (tag: string) => (it.match(new RegExp(`<${tag}>([\\s\\S]*?)</${tag}>`, "i")) || [])[1] || "";
    const title = text(g("title"));
    if (!/^United States\b/i.test(title)) continue;
    const when = Date.parse(text(g("pubDate")));
    if (isNaN(when)) continue;
    let desc = decode(g("description"));
    if (/&#?\w+;/.test(desc)) desc = decode(desc);                   // double-escaped feeds
    const impact = (desc.match(/sprite-(high|medium|low)-impact/i) || [])[1] || "low";
    const rows = desc.split(/<tr/i).slice(1).map(row => [...row.matchAll(/<t[dh][^>]*>([\s\S]*?)<\/t[dh]>/gi)].map(m => text(m[1])));
    const head = rows.find(c => c.some(x => /previous/i.test(x))) || ["Time left", "Impact", "Previous", "Consensus", "Actual"];
    const data = rows.filter(c => c !== head && c.length >= 3).pop() || [];
    const col = (re: RegExp, d: number) => { const i = head.findIndex(x => re.test(x)); return (data[i >= 0 ? i : d] || "").trim(); };
    out.push({
      t: new Date(when).toISOString(),
      title: title.replace(/^United States\s+/i, ""),
      impact: impact[0].toUpperCase() + impact.slice(1).toLowerCase(),
      previous: col(/previous/i, 2), forecast: col(/consensus|forecast/i, 3), actual: col(/actual/i, 4), src: "myfxbook",
    });
  }
  return out;
}

async function fromForexFactory(): Promise<Ev[]> {
  const urls = ["https://nfs.faireconomy.media/ff_calendar_thisweek.json"];
  const out: Ev[] = [];
  for (const u of urls) {
    const r = await fetch(u, { headers: UA });
    if (!r.ok) continue;
    const j = await r.json();
    for (const e of j || []) {
      if (e.country !== "USD") continue;
      const when = Date.parse(e.date);
      if (isNaN(when)) continue;
      out.push({ t: new Date(when).toISOString(), title: e.title, impact: e.impact || "Low",
        previous: e.previous || "", forecast: e.forecast || "", actual: e.actual || "", src: "forexfactory" });
    }
  }
  return out;
}

// ---- Nasdaq economic calendar (has Actual / Consensus / Previous) ----
type NQ = { day: string; name: string; actual: string; consensus: string; previous: string };
const etDay = (t: number) => new Date(t - 4 * 3600e3).toISOString().slice(0, 10);   // US Eastern (EDT) calendar day
const NQCACHE = new Map<string, { at: number; rows: NQ[] }>();
async function nasdaqDay(d: string, liveDay: boolean): Promise<NQ[]> {
  const c = NQCACHE.get(d), ttl = liveDay ? 60_000 : 6 * 3600_000;      // past days are final → keep 6 h
  if (c && Date.now() - c.at < ttl) return c.rows;
  let lastErr: unknown = null;
  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const r = await fetch(`https://api.nasdaq.com/api/calendar/economicevents?date=${d}`,
        { headers: { ...BROWSER, Origin: "https://www.nasdaq.com", Referer: "https://www.nasdaq.com/" } });
      if (!r.ok) throw new Error("nasdaq " + r.status);
      const rows = parseNasdaq(d, await r.json());
      // never let a later, emptier answer erase actuals we already had
      if (c) for (const o of c.rows) if (o.actual && !rows.some(x => x.name === o.name && x.actual)) {
        const x = rows.find(x => x.name === o.name && !x.actual && x.consensus === o.consensus); if (x) x.actual = o.actual; else rows.push(o); }
      NQCACHE.set(d, { at: Date.now(), rows });
      return rows;
    } catch (e) { lastErr = e; await new Promise(r => setTimeout(r, 400)); }
  }
  if (c) return c.rows;                                                  // stale but better than nothing
  throw lastErr;
}
async function fromNasdaq(days: string[]): Promise<NQ[]> {
  const out: NQ[] = [], today = etDay(Date.now());
  const res = await Promise.allSettled(days.map(d => nasdaqDay(d, d > today)));
  let ok = 0;
  res.forEach(r => { if (r.status === "fulfilled") { ok++; out.push(...r.value); } });
  if (!ok) throw new Error("nasdaq unavailable");
  return out;
}
function parseNasdaq(d: string, j: any): NQ[] {
  const out: NQ[] = [];
  {
    const rows = j?.data?.rows || j?.data?.calendar?.rows || [];
    for (const x of rows) {
      const c = String(x.country || x.Country || "");
      if (c && !/united states|^us$|usa/i.test(c)) continue;
      out.push({ day: d, name: String(x.eventName || x.event || x.name || ""), actual: clean(x.actual), consensus: clean(x.consensus ?? x.forecast), previous: clean(x.previous) });
    }
  }
  return out;
}

// ---- Market snapshot for the PDF report (?markets=1): Yahoo chart API, FRED CSV fallback ----
type Q = { last: number; prev: number; chg: number; chg5: number; asOf: string; src: string };
async function yahoo(sym: string): Promise<Q> {
  const r = await fetch(`https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(sym)}?range=1mo&interval=1d`, { headers: BROWSER });
  if (!r.ok) throw new Error("yahoo " + sym + " " + r.status);
  const j = await r.json(), res = j?.chart?.result?.[0];
  const closes: number[] = (res?.indicators?.quote?.[0]?.close || []).filter((v: number | null) => v != null);
  const last = +res?.meta?.regularMarketPrice || closes[closes.length - 1];
  const prev = closes.length >= 2 ? closes[closes.length - 2] : +res?.meta?.chartPreviousClose;
  if (!(last > 0)) throw new Error("yahoo " + sym + " empty");
  const p5 = closes.length >= 6 ? closes[closes.length - 6] : prev;
  return { last, prev, chg: last - prev, chg5: last - p5, asOf: new Date((res?.meta?.regularMarketTime || 0) * 1000).toISOString(), src: "Yahoo" };
}
async function fred(id: string): Promise<Q> {
  const r = await fetch(`https://fred.stlouisfed.org/graph/fredgraph.csv?id=${id}`, { headers: BROWSER });
  if (!r.ok) throw new Error("fred " + id + " " + r.status);
  const rows = (await r.text()).trim().split(/\r?\n/).slice(1).map(l => l.split(",")).filter(x => x[1] && !isNaN(+x[1]));
  if (rows.length < 2) throw new Error("fred " + id + " empty");
  const n = rows.length, last = +rows[n - 1][1], prev = +rows[n - 2][1], p5 = +rows[Math.max(0, n - 6)][1];
  return { last, prev, chg: last - prev, chg5: last - p5, asOf: rows[n - 1][0], src: "FRED" };
}

  ["set", "^SET.BK", "", 1], ["usdthb", "THB=X", "DEXTHUS", 1], ["nasdaq", "^IXIC", "NASDAQCOM", 1],

async function markets() {
  const out: Record<string, Q | null> = {}, errs: string[] = [];
  await Promise.all([
    ...MKT.map(async ([k, y, f, div]) => {
      try { const q = await yahoo(y); if (div !== 1) { q.last /= div; q.prev /= div; q.chg /= div; q.chg5 /= div; } out[k] = q; }
      catch (e) { errs.push(String(e)); if (f) { try { out[k] = await fred(f); } catch (e2) { errs.push(String(e2)); out[k] = null; } } else out[k] = null; }
    }),
    fred("DFII10").then(q => out.real10y = q).catch(e => { errs.push(String(e)); out.real10y = null; }),
    fred("T10YIE").then(q => out.breakeven = q).catch(e => { errs.push(String(e)); out.breakeven = null; }),
  ]);
  return { updated: new Date().toISOString(), markets: out, errors: errs };
}

// first 1-2 sentences of the RSS description, without HTML, "read more" tails or a repeat of the headline
function excerpt(raw: string, title: string): string {
  let e = raw.replace(/<[^>]*>/g, " ").replace(/&[a-z#0-9]+;/gi, " ").replace(/\s+/g, " ").trim();
  e = e.replace(/The post .*? appeared first on .*$/i, "").replace(/(อ่านต่อ|อ่านข่าวต่อ|Read more|Continue reading).*$/i, "").replace(/\[(…|\.\.\.|&#8230;)\]\s*$/, "").trim();
  if (e.startsWith(title)) e = e.slice(title.length).replace(/^[\s\-–:|]+/, "");
  if (e.length < 25 || e === title) return "";
  if (e.length > 300) { const cut = e.slice(0, 300), sp = Math.max(cut.lastIndexOf(". "), cut.lastIndexOf(" ")); e = cut.slice(0, sp > 180 ? sp : 300).trim() + "…"; }
  return e;
}

async function rss(u: string, ms = 7000): Promise<It[]> {
  const r = await fetch(u, { headers: { ...BROWSER, Accept: "application/rss+xml, application/atom+xml, application/xml, text/xml, */*", "Accept-Language": "th-TH,th;q=0.9,en;q=0.8" }, signal: AbortSignal.timeout(ms) });
  if (!r.ok) throw new Error("HTTP " + r.status);
  const x = await r.text(), out: It[] = [];
  const blocks = [...x.matchAll(/<item[\s>]([\s\S]*?)<\/item>/g), ...x.matchAll(/<entry[\s>]([\s\S]*?)<\/entry>/g)];
  if (!blocks.length) throw new Error("no items (" + x.slice(0, 60).replace(/\s+/g, " ") + ")");
  for (const m of blocks) {
    const b = m[1], g = (tag: string) => { const mm = b.match(new RegExp(`<${tag}[^>]*>([\\s\\S]*?)</${tag}>`)); return mm ? text(mm[1].replace(/<!\[CDATA\[([\s\S]*?)\]\]>/g, "$1")) : ""; };
    let t = g("title"), s = g("source") || g("News:Source"), link = g("link");
    if (!link) { const lm = b.match(/<link[^>]*href="([^"]+)"/); link = lm ? decode(lm[1]) : ""; }
    if (!s) { const dm = t.match(/\s+-\s+([^-]{2,40})$/); if (dm && /news\.google|bing\.com/.test(u)) s = dm[1]; }
    if (s && t.endsWith(" - " + s)) t = t.slice(0, -(s.length + 3));
    if (!s) { try { s = new URL(link || u).hostname.replace(/^www\./, ""); } catch { s = ""; } }
    if (/bing\.com/.test(link)) { const um = link.match(/[?&]url=([^&]+)/); if (um) link = decodeURIComponent(um[1]); }
    const d = new Date(g("pubDate") || g("published") || g("updated") || g("dc:date") || Date.now());
    const x = excerpt(g("description") || g("content:encoded") || g("summary") || g("content"), t);
    if (t) out.push({ t, s, u: link, x, d: isNaN(+d) ? new Date().toISOString() : d.toISOString(), lang: /[฀-๿]/.test(t) ? "th" : "en" });
  }
  return out;
}
const TR = new Map<string, string>();   // translations survive between requests (saves the free quota)
let trBlockedUntil = 0;
async function translate(items: It[], errs: string[]) {
  // texts to translate: English headlines, plus the excerpts of the top stories
  const want: [It, "tt" | "xt", string][] = [];
  for (const it of items.filter(i => i.lang === "en")) {
    const a = TR.get(it.t), b = it.x ? TR.get(it.x) : undefined;
    if (a) it.tt = a; else want.push([it, "tt", it.t]);
    if (it.x) { if (b) it.xt = b; else want.push([it, "xt", it.x]); }
  }
  if (!want.length || Date.now() < trBlockedUntil) return;
  const batch = want.slice(0, 70);
  try {   // one POST for everything: fewer calls, fewer 429s
    const r = await fetch("https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=th&dt=t", {
      method: "POST", headers: { ...BROWSER, "Content-Type": "application/x-www-form-urlencoded;charset=UTF-8" },
      body: "q=" + encodeURIComponent(batch.map(w => w[2].replace(/\n/g, " ")).join("\n")), signal: AbortSignal.timeout(9000) });
    if (!r.ok) { if (r.status === 429) trBlockedUntil = Date.now() + 10 * 60e3; throw new Error("HTTP " + r.status); }
    const j = await r.json(), out = (j[0] || []).map((s: unknown[]) => String(s[0] || "")).join("").split("\n");
    batch.forEach(([it, f, src], k) => { const v = (out[k] || "").trim(); if (v && /[\u0E00-\u0E7F]/.test(v)) { it[f] = v; TR.set(src, v); } });
  } catch (e) { errs.push("translate: " + e); }
}

const SB = Deno.env.get("SUPABASE_URL") || "", SRK = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") || "";
const sbH = { apikey: SRK, Authorization: "Bearer " + SRK, "Content-Type": "application/json" };
async function trKey(t: string) { const h = new Uint8Array(await crypto.subtle.digest("SHA-1", new TextEncoder().encode(t.trim()))); return [...h].slice(0, 10).map(b => b.toString(16).padStart(2, "0")).join(""); }
function enTexts(body: Record<string, Record<string, It[]>>): string[] {
  const out: string[] = []; for (const r of Object.values(body)) for (const l of Object.values(r)) for (const x of l) if (x.lang === "en") { out.push(x.t); if (x.x) out.push(x.x); } return out; }
async function trLoad(body: Record<string, Record<string, It[]>>, errs: string[]) {
  if (!SB || !SRK) return;
  const texts = [...new Set(enTexts(body))].filter(t => !TR.has(t)); if (!texts.length) return applyTr(body);
  const keys = new Map<string, string>(); for (const t of texts) keys.set(await trKey(t), t);
  try {
    const r = await fetch(`${SB}/rest/v1/news_tr?select=k,th&k=in.(${[...keys.keys()].join(",")})`, { headers: sbH, signal: AbortSignal.timeout(5000) });
    if (!r.ok) throw new Error("HTTP " + r.status + " " + (await r.text()).slice(0, 80));
    for (const row of await r.json() as { k: string; th: string }[]) { const t = keys.get(row.k); if (t) TR.set(t, row.th); }
  } catch (e) { errs.push("news_tr read: " + e); }
  applyTr(body);
}
function applyTr(body: Record<string, Record<string, It[]>>) {
  for (const r of Object.values(body)) for (const l of Object.values(r)) for (const x of l) if (x.lang === "en") {
    if (!x.tt && TR.has(x.t)) x.tt = TR.get(x.t); if (x.x && !x.xt && TR.has(x.x)) x.xt = TR.get(x.x); } }
async function trSave(pairs: [string, string][]) {
  const feedBody = (FEED?.body as { feed?: Record<string, Record<string, It[]>> })?.feed; if (!feedBody || !SB || !SRK) return 0;
  const known = new Set(enTexts(feedBody)), rows: { k: string; th: string }[] = [];
  for (const [src, th] of pairs.slice(0, 120)) {
    if (typeof src !== "string" || typeof th !== "string" || !known.has(src) || th.length > 1500 || !/[\u0E00-\u0E7F]/.test(th)) continue;
    if (TR.has(src)) continue;   // first translation wins; nobody can overwrite one already stored
    TR.set(src, th.trim()); rows.push({ k: await trKey(src), th: th.trim() });
  }
  if (!rows.length) return 0;
  await fetch(`${SB}/rest/v1/news_tr`, { method: "POST", headers: { ...sbH, Prefer: "resolution=ignore-duplicates,return=minimal" }, body: JSON.stringify(rows) });
  applyTr(feedBody); return rows.length;
}
const norm = (t: string) => t.toLowerCase().replace(/[^฀-๿a-z0-9]/g, "").slice(0, 60);
async function feed() {
  const errs: string[] = [], cnt: Record<string, number | string> = {}, now = Date.now(), MAXAGE = 3 * 86400e3;
  const got = await Promise.all(SOURCES.map(src => rss(src.u).then(a => { cnt[src.id] = a.length; return a.map(x => ({ x: { ...x, s: src.s && !/news\.google|bing/.test(src.u) ? src.s : x.s }, src })); })
    .catch(e => { cnt[src.id] = "✗ " + String(e).replace(/^Error: /, "").slice(0, 80); errs.push(src.id + ": " + e); return []; })));
  const ORDER = ["gold", "tech", "geo", "econ", "trend"];
  const body: Record<string, Record<string, It[]>> = { intl: {}, th: {} }, seen = new Set<string>();
  const bucket: Record<string, It[]> = {};
  for (const { x, src } of got.flat()) {
    if (now - +new Date(x.d) > MAXAGE || +new Date(x.d) > now + 3600e3) continue;
    const k = norm(x.t); if (!k || seen.has(k)) continue; seen.add(k);
    const reg = src.reg || (x.lang === "en" ? "intl" : RX.gold.test(x.t) ? (/โลก|spot|ดอลลาร์|ออนซ์|xau/i.test(x.t) && !THAI.test(x.t) ? "intl" : "th") : WORLD.test(x.t) && !THAI.test(x.t) ? "intl" : "th");
    let cat = src.cat || "";
    if (!cat || cat === "trend") { const c = ORDER.find(c => c !== "trend" && RX[c].test(x.t)); cat = src.cat === "trend" ? (c === "gold" ? "gold" : "trend") : (c || "trend"); }
    if (cat === "gold" && !RX.gold.test(x.t)) cat = "econ";
    (bucket[reg + "." + cat] ||= []).push(x);
  }
  for (const reg of ["intl", "th"]) for (const cat of ORDER)
  {   // Thai-language stories first (up to 9), English fills the rest
    const a = (bucket[reg + "." + cat] || []).sort((a, b) => +new Date(b.d) - +new Date(a.d));
    const th = a.filter(x => x.lang === "th").slice(0, 9), en = a.filter(x => x.lang !== "th");
    body[reg][cat] = [...th, ...en].slice(0, 14);
  }
  await trLoad(body, errs);
  await translate(Object.values(body.intl).flat(), errs);
  return { updated: new Date().toISOString(), feed: body, counts: cnt, errors: errs };
}

const words = (s: string) => { let t = " " + s.toLowerCase() + " "; for (const [re, v] of SYN) t = t.replace(re, " " + v + " ");
  return new Set(t.replace(/[^a-z ]/g, " ").split(/\s+/).filter(w => w.length > 1 && !STOP.has(w))); };
const sim = (a: string, b: string) => { const A = words(a), B = words(b); let n = 0; A.forEach(w => { if (B.has(w)) n++; });
  // period tokens must agree when both specify one
  for (const p of ["mm", "yy", "qq"]) if ((A.has(p) || B.has(p)) && A.has(p) !== B.has(p) && [...A, ...B].some(w => ["mm", "yy", "qq"].includes(w) && w !== p)) return 0;
  if (A.has("core") !== B.has("core")) return 0;
  return n / Math.max(1, Math.min(A.size, B.size)); };
